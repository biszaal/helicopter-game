# Coins (Roadmap Phase 6) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Collectible coins — safe trails worth 1 and graze-band risk clusters worth 5 — with a generous pickup, pop + rising-pitch ding, HUD counter, and a persistent wallet.

**Architecture:** Coins are a pooled world object exactly like rocks: pure placement geometry in `spawning.ts` (TDD), a `useCoinPool` of shared-value slots, spawn/collect/retire inside the fixed-timestep loop with zero allocation, render as parked-when-idle Skia circles, JS-side effects via `useAnimatedReaction` + `runOnJS`. Persistence extends `settings.tsx`.

**Tech Stack:** existing stack only — reanimated worklets, RN Skia, expo-audio (via `restartAndPlay`), AsyncStorage, vitest.

## Global Constraints

- Spec: `docs/superpowers/specs/2026-07-06-coins-design.md`. Gameplay tunables in `config.coins` + `config.events.coinPopSeconds = 0.4`; cosmetics in `theme.coins`.
- Zero allocation in the loop; pooled slots; engine zeroes/retires event state (pop window) so render deriveds idle.
- ONE shared color set — never derive colors per coin instance (perf lesson from 2026-07-04).
- Rate changes only on one-shot players (ding OK; never on loops). All plays go through `restartAndPlay`.
- Trail coins must pass the occupied-envelope fitness check (drift envelopes included); **risk clusters skip it** — they deliberately sit in the graze band, and their pickup box overlaps the parent obstacle's rect by design.
- Wallet coins bank on EVERY death (the current `recordRun` early-returns on non-best runs — must be restructured, not just extended).
- Verification: `npx tsc --noEmit` + `npx vitest run` per task; device checks by the user in Expo Go.

```ts
export interface CoinPoint {
  x: number;
  y: number;
}

/**
 * Trail template: `count` coin centres, `spacing` apart in x (relative to
 * the trail start), shaped by `kind` (0 line, 1 full sine cycle, 2 arc
 * bump toward the ceiling). y clamps to [minY, maxY] so a big amplitude
 * near a wall flattens instead of clipping.
 */
export function coinTrailPoints(
  count: number,
  spacing: number,
  centerY: number,
  amp: number,
  kind: number,
  minY: number,
  maxY: number,
): CoinPoint[] {
  'worklet';
  const pts: CoinPoint[] = [];
  for (let i = 0; i < count; i++) {
    const t = count > 1 ? i / (count - 1) : 0;
    let y = centerY;
    if (kind === 1) y = centerY + amp * Math.sin(2 * Math.PI * t);
    else if (kind === 2) y = centerY - amp * 4 * t * (1 - t);
    pts.push({ x: i * spacing, y: Math.min(maxY, Math.max(minY, y)) });
  }
  return pts;
}

/** True when a coin's pickup box (half-size `half`) misses every blocker. */
export function coinFits(
  cx: number,
  cy: number,
  half: number,
  occupied: OccupiedRect[],
): boolean {
  'worklet';
  for (let i = 0; i < occupied.length; i++) {
    const r = occupied[i];
    if (
      cx + half > r.x1 &&
      cx - half < r.x2 &&
      cy + half > r.y1 &&
      cy - half < r.y2
    ) {
      return false;
    }
  }
  return true;
}

/** Risk trio beside a rock's right edge — collecting them IS the graze line. */
export function rockClusterPoints(
  rockRight: number,
  centerY: number,
  halfH: number,
  offset: number,
): CoinPoint[] {
  'worklet';
  const x = rockRight + offset;
  return [
    { x, y: centerY - halfH },
    { x, y: centerY },
    { x, y: centerY + halfH },
  ];
}

/** Risk trio down a rock-wall gate, spanning half the gap. */
export function gateClusterPoints(
  x: number,
  gapCenter: number,
  gapHalf: number,
): CoinPoint[] {
  'worklet';
  return [
    { x, y: gapCenter - gapHalf * 0.5 },
    { x, y: gapCenter },
    { x, y: gapCenter + gapHalf * 0.5 },
  ];
}

/** Risk trio along a pinch corridor centre at 30/50/70% of the span. */
export function pinchClusterPoints(
  worldX: number,
  length: number,
  corridorCenter: number,
): CoinPoint[] {
  'worklet';
  return [
    { x: worldX + length * 0.3, y: corridorCenter },
    { x: worldX + length * 0.5, y: corridorCenter },
    { x: worldX + length * 0.7, y: corridorCenter },
  ];
}
```

import { useSharedValue, type SharedValue } from 'react-native-reanimated';
import { config } from '../config';

export interface CoinSlot {
  active: SharedValue<number>;
  worldX: SharedValue<number>;
  centerY: SharedValue<number>;
  /** Wallet value (1 trail, 5 risk) — also picks the risk look. */
  value: SharedValue<number>;
  /** Wall clock (ms) when collected; 0 while collectable. Drives the pop. */
  collectedAtMs: SharedValue<number>;
  /** Cosmetic sparkle variety. */
  seed: SharedValue<number>;
}

/** Fixed-size pool of coin slots — reused for the whole app lifetime. */
export function useCoinPool(): CoinSlot[] {
  const slots: CoinSlot[] = [];
  for (let i = 0; i < config.coins.poolSize; i++) {
    // eslint-disable-next-line react-hooks/rules-of-hooks -- fixed count
    slots.push({
      active: useSharedValue(0),
      worldX: useSharedValue(0),
      centerY: useSharedValue(0),
      value: useSharedValue(1),
      collectedAtMs: useSharedValue(0),
      seed: useSharedValue(0),
    });
  }
  return slots;
}

  const collectOccupied = (): OccupiedRect[] => {
    'worklet';
    const occupied: OccupiedRect[] = [];
    for (let i = 0; i < rocks.length; i++) {
      const r = rocks[i];
      if (r.active.value === 0) continue;
      occupied.push({

```ts
  const coins = useCoinPool();
  const nextTrailAt = useSharedValue<number>(config.coins.trail.firstDelaySeconds);
  const runCoins = useSharedValue(0);

  const claimCoin = (x: number, y: number, value: number) => {
    'worklet';
    for (let i = 0; i < coins.length; i++) {
      const c = coins[i];
      if (c.active.value === 0) {
        c.worldX.value = x;
        c.centerY.value = y;
        c.value.value = value;
        c.collectedAtMs.value = 0;
        c.seed.value = Math.random();
        c.active.value = 1;
        return;
      }
    }
  };

  /** Risk trio: deliberate graze-band placement — no fitness check. */
  const spawnRiskCluster = (pts: CoinPoint[]) => {
    'worklet';
    for (let i = 0; i < pts.length; i++) {
      claimCoin(pts[i].x, pts[i].y, config.coins.riskCluster.value);
    }
  };

  const spawnTrail = (scroll: number) => {
    'worklet';
    const t = config.coins.trail;
    const count =
      t.countMin + Math.floor(Math.random() * (t.countMax - t.countMin + 1));
    const amp = t.ampMin + Math.random() * (t.ampMax - t.ampMin);
    const kind = Math.floor(Math.random() * 3);
    const r = config.coins.radius;
    const minY = config.world.wallThickness + 2 * r;
    const maxY = 1 - config.world.wallThickness - 2 * r;
    const centerY = minY + amp + Math.random() * (maxY - minY - 2 * amp);
    const pts = coinTrailPoints(count, t.spacing, centerY, amp, kind, minY, maxY);
    const startX = scroll + aspectRatio + r;
    const half = r * config.coins.pickupScale + config.coins.placementMargin;
    const occupied = collectOccupied();
    for (let i = 0; i < pts.length; i++) {
      const x = startX + pts[i].x;
      if (coinFits(x, pts[i].y, half, occupied)) claimCoin(x, pts[i].y, t.value);
    }
  };

  /** A late obstacle landing on parked coins deletes them — a coin inside
   * a rock is a lie (fairness rule, spec). */
  const clearCoinsInRect = (x1: number, x2: number, y1: number, y2: number) => {
    'worklet';
    const r = config.coins.radius;
    for (let i = 0; i < coins.length; i++) {
      const c = coins[i];
      if (c.active.value === 0 || c.collectedAtMs.value !== 0) continue;
      if (
        c.worldX.value + r > x1 &&
        c.worldX.value - r < x2 &&
        c.centerY.value + r > y1 &&
        c.centerY.value - r < y2
      ) {
        c.active.value = 0;
      }
    }
  };
```

```tsx
import { Circle, Group } from '@shopify/react-native-skia';
import { useDerivedValue, type SharedValue } from 'react-native-reanimated';
import { config } from '../config';
import { type CoinSlot } from '../engine/useCoinPool';
import { theme } from './theme';

interface CoinProps {
  slot: CoinSlot;
  screenHeight: number;
  renderScrollX: SharedValue<number>;
  wallClockMs: SharedValue<number>;
}

/**
 * One pooled coin: gold disc (bigger + amber when it's a risk coin), and a
 * scale-up-and-fade pop when collected — a pure function of
 * wallClockMs − collectedAtMs, retired by the engine's pop window.
 */
export function Coin({ slot, screenHeight, renderScrollX, wallClockMs }: CoinProps) {
  const rPx = config.coins.radius * screenHeight;
  const popSeconds = config.events.coinPopSeconds;

  const transform = useDerivedValue(() => {
    if (slot.active.value === 0) return [{ translateX: -2 * screenHeight }];
    const sizeScale = slot.value.value > 1 ? theme.coins.riskSizeScale : 1;
    let pop = 1;
    if (slot.collectedAtMs.value !== 0) {
      const t = (wallClockMs.value - slot.collectedAtMs.value) / 1000 / popSeconds;
      pop = 1 + (theme.coins.popScale - 1) * Math.min(1, Math.max(0, t));
    }
    return [
      { translateX: (slot.worldX.value - renderScrollX.value) * screenHeight },
      { translateY: slot.centerY.value * screenHeight },
      { scale: sizeScale * pop },
    ];
  });
  const opacity = useDerivedValue(() => {
    if (slot.active.value === 0) return 0;
    if (slot.collectedAtMs.value === 0) return 1;
    const t = (wallClockMs.value - slot.collectedAtMs.value) / 1000 / popSeconds;
    return Math.max(0, 1 - t);
  });
  const color = useDerivedValue(() =>
    slot.value.value > 1 ? theme.colors.amber : theme.coins.gold,
  );

  return (
    <Group transform={transform} opacity={opacity}>
      <Circle cx={0} cy={0} r={rPx} color={color} />
      <Circle
        cx={-rPx * 0.3}
        cy={-rPx * 0.3}
        r={rPx * 0.35}
        color={theme.coins.highlight}
      />
    </Group>
  );
}
```

- [ ] **Step 2: `playCoinDing`.** In `sounds.ts`: add `ding: AudioPlayer` to the record, create with `PLAYER_OPTIONS`, `ding.volume = 0.6;`, and:

```ts
const DING_STREAK_WINDOW_MS = 1000;
const DING_RATE_STEP = 0.06;
const DING_RATE_MAX = 1.6;
let dingStreak = 0;
let lastDingAtMs = 0;

/** Coin collected. Consecutive pickups within a second climb in pitch —
 * a trail reads as an arpeggio. Rate on a ONE-SHOT player is safe (the
 * looper lesson applies to loops). */
export function playCoinDing(): void {
  if (!soundOn || players === null) return;
  const now = Date.now();
  dingStreak = now - lastDingAtMs <= DING_STREAK_WINDOW_MS ? dingStreak + 1 : 0;
  lastDingAtMs = now;
  players.ding.setPlaybackRate(
    Math.min(DING_RATE_MAX, 1 + DING_RATE_STEP * dingStreak),
  );
  restartAndPlay(players.ding);
}
```
